\section{Syntactic definition of pure type systems}
\label{app:pts-syntax}

\subsection{Specifications}
\begin{definition}
  A \textit{\ac{PTS} specification} consists of the following data:
  \begin{enumerate}
  \item A set $\St$ of \textit{sorts}, denoted $\tmcst s$;
  \item A relation $\Ax \subseteq \St \times \St$ of \textit{axioms}, denoted $\tmcst s_1 : \tmcst s_2$;
  \item A relation $\Ru \subseteq \St \times \St \times \St$ of \textit{rules} for function types.
  \end{enumerate}
\end{definition}

\subsection{Syntax}

\[
\begin{array}{lrcl}
  \mbox{Expressions}       & M,N,A,B,C      & ::=  & x_i \mid \tmcst s \mid \tppi A B \mid \tmlam M \mid \tmapp M N \mid \tmclo \sigma M \\
  \mbox{Contexts}          & \Gamma, \Delta & ::=  & \ctxempty \mid \ctxext\Gamma A \\
  \mbox{Substitutions}     & \sigma, \tau   & ::=  & \subempty \mid\subid \mid \subwk \mid \subext \sigma M \mid \subclo \sigma \tau
\end{array}
\]

\subsubsection{Operations on contexts}

\noindent
$\boxed{\ctxlookup \Gamma {x_i} A}$ -- Context lookup
\[
\begin{array}{c}
  \infer{
    \ctxlookup {\ctxext \Gamma A} {x_0} {\tmclo \subwk A}
  }{
    \wfctx {\ctxext \Gamma A}
  }
  \qquad
  \infer{
    \ctxlookup {\ctxext \Gamma B} {x_{i+1}} {\tmclo \subwk A}
  }{
    \ctxlookup \Gamma {x_i} A
    &
    \wfctx {\ctxext \Gamma B}
  }
\end{array}
\]

\subsection{Judgments}

\subsubsection{Well-formedness judgments}

\noindent
$\boxed{\wfctx \Gamma}$ -- $\Gamma$ is a well-formed context.
\[
\begin{array}{c}
  \infer{
    \wfctx \ctxempty
  }{
    % Axiom
  }
  \qquad
  \infer{
    \wfctx {\ctxext \Gamma A}
  }{
    \wfctx \Gamma
    &
    \wfexp \Gamma A {\tmcst s}
  }
\end{array}
\]


\noindent
$\boxed{\wfexp \Gamma M A}$ -- Expression $M$ has type $A$ in context $\Gamma$.
\[
\begin{array}{c}
  \infer{
    \wfexp \Gamma {x_i} A
  }{
    \ctxlookup \Gamma {x_i} A
  }
  \qquad
  \infer{
    \wfexp \Gamma {\tmcst s_1} {\tmcst s_2}
  }{
    \wfctx \Gamma
    &
    (\tmcst s_1 : \tmcst s_2) \in \Ax
  }
  \qquad
  \infer{
    \wfexp \Gamma {\tppi A B} {\tmcst s_3}
  }{
    \wfexp \Gamma A {\tmcst s_1}
    &
    \wfexp {\ctxext \Gamma A} B {\tmcst s_2}
    &
    (\tmcst s_1, \tmcst s_2, \tmcst s_3) \in \Ru
  }
  \\[8pt]
  \infer{
    \wfexp \Gamma {\tmclo \sigma M} {\tmclo \sigma A}
  }{
    \wfexp \Delta M A
    &
    \wfsub \Gamma \sigma \Delta
  }
  \qquad
  \infer{
    \wfexp \Gamma {\tmlam M} {\tppi A B}
  }{
    \wftyp \Gamma \tppi A B
    &
    \wfexp {\ctxext \Gamma A} M B
  }
  \qquad
  \infer{
    \wfexp \Gamma {\tmapp M N} {\tmclo {\subext \subid N} B}
  }{
    \wfexp \Gamma M {\tppi A B}
    &
    \wfexp \Gamma N A
  }
  \\[8pt]
  \infer{
    \wfexp \Gamma M B
  }{
    \wfexp \Gamma M A
    &
    \eqtyp \Gamma A B
  }
\end{array}
\]


\noindent
$\boxed{\wftyp \Gamma A}$ -- $A$ is a well-formed type in context $\Gamma$.
\[
\begin{array}{c}
  \infer{
    \wftyp \Gamma {\tmcst s}
  }{
    \wfctx \Gamma
    &
    \text{there is no axiom } \tmcst s : \tmcst s' \in \Ax
  }
  \qquad
  \infer{
    \wftyp \Gamma A
  }{
    \wfexp \Gamma A {\tmcst s}
  }
\end{array}
\]


\noindent
$\boxed{\wfsub \Gamma \sigma \Delta}$ -- $\sigma$ is a well-formed substitution from $\Delta$ to $\Gamma$.
\[
\begin{array}{c}
  \infer{
    \wfsub \Gamma \subempty \ctxempty
  }{
    \wfctx \Gamma
  }
  \qquad
  \infer{
    \wfsub \Gamma \subid \Gamma
  }{
    \wfctx \Gamma
  }
  \qquad
  \infer{
    \wfsub {\ctxext \Gamma A} \subwk \Gamma
  }{
    \wfctx {\ctxext \Gamma A}
  }
  \qquad
  \infer{
    \wfsub \Gamma {\subext \sigma M} {\ctxext \Delta A}
  }{
    \wfsub \Gamma \sigma \Delta
    &
    \wfexp \Delta A {\tmcst s}
    &
    \wfexp \Gamma M {\tmclo \sigma A}
  }
  \\[8pt]
  \infer{
    \wfsub {\Gamma_1} {\subclo \sigma \tau} {\Gamma_3}
  }{
    \wfsub {\Gamma_1} \sigma {\Gamma_2}
    &
    \wfsub {\Gamma_2} \tau {\Gamma_3}
  }
  \qquad
  \infer{
    \wfsub \Gamma \sigma {\Delta'}
  }{
    \wfsub \Gamma \sigma \Delta
    &
    \eqctx \Delta {\Delta'}
  }
\end{array}
\]

\subsubsection{Equality judgments}

\noindent
$\boxed{\eqctx \Gamma \Delta}$ -- $\Gamma$ and $\Delta$ are equivalent contexts.
\[
\begin{array}{c}
  \infer{
    \eqctx \cdot \cdot
  }{
    % Axiom
  }
  \qquad
  \infer{
    \eqctx {\ctxext \Gamma A} {\ctxext \Delta B}
  }{
    \eqctx \Gamma \Delta
    &
    \eqexp \Gamma A B {\tmcst s}
  }
\end{array}
\]



\noindent
$\boxed{\eqexp \Gamma M N A}$ -- $M$ and $N$ are equivalent expressions of type $A$ in context $\Gamma$.

\noindent
$\beta$-reduction and $\eta$-expansion
\[
\begin{array}{c}
  \infer{
    \eqexp \Gamma {\tmapp {(\tmlam M)} N} {\tmclo {\subext \subid N} M} {\tmclo {\subext \subid N} B}
  }{
    \wfexp {\ctxext \Gamma A} M B
    &
    \wfexp \Gamma N A
  }
  \qquad
  \infer{
    \eqexp \Gamma M {\tmlam {(\tmapp {\tmclo \subwk M} {x_0})}} {\tppi A B}
  }{
    \wfexp \Gamma M {\tppi A B}
  }
\end{array}
\]

\noindent
Substitution propagation
\[
\begin{array}{c}
\infer{
    \eqexp \Gamma {\tmclo \sigma {\tmcst s}} {\tmcst s} {\tmcst s'}
  }{
    \wfexp \Delta {\tmcst s} {\tmcst s'}
    &
    \wfsub \Gamma \sigma \Delta
  }
  \qquad
  \infer{
    \eqexp \Gamma {\tmclo {\subext \sigma M} {x_0}} M {\tmclo \sigma A}
  }{
    \wfsub \Gamma {\subext \sigma M} {\ctxext \Delta A}
  }
  \qquad
  \infer{
    \eqexp \Gamma {\tmclo {\subext \sigma M} {x_{i+1}}} {\tmclo \sigma x_i} {\tmclo \sigma B}
  }{
    \wfsub \Gamma {\subext \sigma M} {\ctxext \Delta A}
    &
    \ctxlookup \Delta {x_i} B
  }
  \\[8pt]
  \infer{
    \eqexp \Gamma {\tmclo \subid M} M A
  }{
    \wfexp \Gamma M A
  }
  \qquad
  \infer{
    \eqexp \Gamma {\tmclo \sigma {\tppi A B}} {\tppi {\tmclo \sigma A} {\tmclo {\subext {\subclo \subwk \sigma} {x_0}} B}} {\tmcst s}
  }{
    \wfsub \Gamma \sigma \Delta
    &
    \wfexp \Delta {\tppi A B} {\tmcst s}
  }
  \qquad
  \infer{
    \eqexp \Gamma {\tmclo {\subclo \sigma \tau} M} {\tmclo {\sigma} {\tmclo \tau M}} {\tmclo {\subclo \sigma \tau} A}
  }{
    \wfsub \Gamma \sigma \Delta
    &
    \wfsub \Delta \tau {\Delta'}
    &
    \wfexp {\Delta'} M A
  }
  \\[8pt]
  \infer{
    \eqexp \Gamma {\tmclo \sigma {\tmlam M}} {\tmlam {\tmclo {\subclo \subwk \sigma} M}} {\tmclo \sigma {\tppi A B}}
  }{
    \wfsub \Gamma \sigma \Delta
    &
    \wfexp {\ctxext \Delta A} M B
  }
  \qquad
  \infer{
    \eqexp \Gamma {\tmclo \sigma {(\tmapp M N)}} {\tmapp {\tmclo \sigma M} {\tmclo \sigma N}} {\tmclo {\subext \sigma {\tmclo \sigma N}} B}
  }{
    \wfsub \Gamma \sigma \Delta
    &
    \wfexp \Delta M {\tppi A B}
    &
    \wfexp \Delta N A
  }
\end{array}
\]

\noindent
Equivalence relation rules
\[
\begin{array}{c}
  \infer{
    \eqexp \Gamma M M A
  }{
    \wfexp \Gamma M A
  }
  \qquad
  \infer{
    \eqexp \Gamma N M A
  }{
    \eqexp \Gamma M N A
  }
  \qquad
  \infer{
    \eqexp \Gamma {M_1} {M_3} A
  }{
    \eqexp \Gamma {M_1} {M_2} A
    &
    \eqexp \Gamma {M_2} {M_3} A
  }
\end{array}
\]


\noindent
Congurence rules
\[
\begin{array}{c}
  \infer{
    \eqexp \Gamma {\tppi A B} {\tppi {A'} {B'}} {\tmcst s_3}
  }{
    (\tmcst s_1, \tmcst s_2, \tmcst s_3) \in \Ru
    &
    \eqexp \Gamma A {A'} {\tmcst s_1}
    &
    \eqexp {\ctxext \Gamma A} B {B'} {\tmcst s_2}
  }
  \qquad
  \infer{
    \eqexp \Gamma {\tmlam M} {\tmlam {M'}} {\tppi A B}
  }{
    \wftyp \Gamma {\tppi A B}
    &
    \eqexp {\ctxext \Gamma A} M {M'} B
  }
  \\[8pt]
  \infer{
    \eqexp \Gamma {\tmapp M N} {\tmapp {M'} {N'}} {\tmclo {\subext \subid N} B}
  }{
    \eqexp \Gamma M {M'} {\tppi A B}
    &
    \eqexp \Gamma N {N'} A
  }
  \qquad
  \infer{
    \eqexp \Gamma {\tmclo \sigma M} {\tmclo {\sigma'} {M'}} {\tmclo \sigma A}
  }{
    \eqsub \Gamma \sigma {\sigma'} \Delta
    &
    \eqexp \Delta M {M'} A
  }
\end{array}
\]

\noindent
$\boxed{\eqtyp \Gamma A B}$ -- $A$ and $B$ are equivalent types in context $\Gamma$.
\[
\begin{array}{c}
  \infer{
    \eqtyp \Gamma {\tmcst s} {\tmcst s}
  }{
    \wfctx \Gamma
    &
    \text{there is no axiom } \tmcst s : \tmcst s' \in \Ax
  }
  \qquad
  \infer{
    \eqtyp \Gamma A B
  }{
    \eqexp \Gamma A B {\tmcst s}
  }
\end{array}
\]

Congruence rules
\[
\begin{array}{c}
  \infer{
    \eqsub \Gamma {\subext \sigma M} {\subext {\sigma'} {M'}} {\ctxext \Delta A}
  }{
    \eqsub \Gamma \sigma {\sigma'} \Delta
    &
    \wftyp \Delta A
    &
    \eqexp \Gamma M {M'} {\tmclo \sigma A}
  }
  \qquad
  \infer{
    \eqsub {\Gamma_1} {\subclo \sigma \tau} {\subclo {\sigma'} {\tau'}} {\Gamma_3}
  }{
    \eqsub {\Gamma_1} \sigma {\sigma'} {\Gamma_2}
    &
    \eqsub {\Gamma_2} \tau {\tau'} {\Gamma_3}
  }
\end{array}
\]


\subsubsection{Properties of judgments}

\begin{lemma}[Presupposition]
  Components of judgments are always well-formed.
  Specifically:
  \begin{enumerate}
  \item If $\wfexp \Gamma M A$, then $\wfctx \Gamma$ and $\wftyp \Gamma A$.
  \item If $\wfsub \Gamma \sigma \Delta$, $\wfctx \Gamma$ and $\wfctx \Delta$.
  \item If $\eqctx \Gamma \Delta$, then $\wfctx \Gamma$ and $\wfctx \Delta$.
  \item If $\eqexp \Gamma M N A$, then $\wfctx \Gamma$, $\wftyp \Gamma A$, $\wfexp \Gamma M A$, and $\wfexp \Gamma N A$.
  \item If $\eqsub \Gamma \sigma \tau \Delta$, then $\wfctx \Gamma$, $\wfctx \Delta$, $\wfsub \Gamma \sigma \Delta$, and $\wfsub \Gamma \tau \Delta$.
  \end{enumerate}
\end{lemma}
\begin{proof}
  By induction on the given derivation.
\end{proof}
